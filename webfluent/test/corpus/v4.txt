    conflict(problem: Problem) on 409
    http(error: HttpError)
}
store People {
    action invite(name: String) -> User ! CreateUserError {
        let user = try await Backend.create(body: name)
        return user
    }
    action submit(name: String) {
        match await Backend.create(body: name) {
            .ok(u) { said = u.name }
            .err(.http(.status(503, _))) { said = "down" }
            .err(_) { said = "failed" }
        }
    }
}
